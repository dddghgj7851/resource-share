export default { async fetch(request, env, ctx) { const url = new URL(request.url); // 1. 爱发电 Webhook 接收端 if (url.pathname === "/afdian-webhook") { // ★★★ 关键调试：强制打印所有收到的请求头 ★★★ console.log("=== Incoming Request Headers ==="); for (const [key, value] of request.headers.entries()) { console.log(`${key}: ${value}`); } // 如果是 GET 请求（爱发电用来探活），直接返回 ok if (request.method === "GET") { return new Response("ok", { status: 200 }); } // 必须是 POST 请求 if (request.method !== "POST") { return new Response(JSON.stringify({ ec: 405, em: "Method Not Allowed" }), { status: 405, headers: { "Content-Type": "application/json" } }); } try { // 读取 Body const bodyText = await request.text(); console.log("=== Request Body ==="); console.log(bodyText); // 打印原始 JSON 字符串 // 尝试解析 JSON const data = JSON.parse(bodyText); // ★★★ 关键逻辑：不管签名对不对，先强制写入 KV 看看能不能解开锁 ★★★ // 提取订单号 (尝试常见字段) const order_id = data.data?.order?.order_id || data.order_id || "UNKNOWN_ID"; // 提取资源 ID (尝试从商品备注中找) let resId = "default"; try { const remark = data.data?.order?.remark || ""; const match = remark.match(/ID:\s*(\w+)/i); if (match && match[1]) { resId = match[1]; } } catch(e) {} // 生成 Token (取订单号后8位) const token = order_id.slice(-8).toUpperCase(); // 强制写入 KV await env.RESOURCE_KV.put(`token:${token}`, resId); console.log(`[SUCCESS] KV Written: token:${token} -> ${resId}`); // 返回成功给爱发电 return new Response(JSON.stringify({ ec: 200, em: "Success" }), { headers: { "Content-Type": "application/json" } }); } catch (e) { console.error("!!! PROCESSING ERROR !!!", e); return new Response(JSON.stringify({ ec: 500, em: e.message }), { status: 500 }); } } return new Response("Not Found", { status: 404 }); } };
